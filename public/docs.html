🚀 Exemplos Rápidos de Integração

Exemplo de verificação de licença via endpoint POST /api/v1/licenses/verify: 

const response = await fetch(`${window.location.origin}/api/v1/licenses`, {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer sk_live_SUA_API_KEY'
  },
  body: JSON.stringify({
    clientName: 'João da Silva',
    clientEmail: 'joao@cliente.com',
    extensionId: 'youtube', // dszap, sender, tele, email, etc.
    validityDays: 365
  })
});
const data = await response.json();
console.log('Chave gerada:', data.data.license_key);
import requests

url = f"{window.location.origin}/api/v1/licenses"
headers = {
    "Content-Type": "application/json",
    "Authorization": "Bearer sk_live_SUA_API_KEY"
}
payload = {
    "clientName": "João da Silva",
    "clientEmail": "joao@cliente.com",
    "extensionId": "youtube",
    "validityDays": 365
}
res = requests.post(url, json=payload, headers=headers)
print(res.json())
curl -X POST https://api-licencas-minhas-ferramentas.vercel.app/api/v1/licenses \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer sk_live_SUA_API_KEY" \
  -d '{
    "clientName": "João da Silva",
    "clientEmail": "joao@cliente.com",
    "extensionId": "youtube",
    "validityDays": 365
  }'
const response = await fetch(`${window.location.origin}/api/v1/licenses/verify`, {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer sk_live_SUA_API_KEY'
  },
  body: JSON.stringify({
    licenseKey: 'CHAVE-DA-LICENCA',
    extensionId: 'youtube'
  })
});
const data = await response.json();
console.log('Licença válida?', data.data?.valid);
