🚀 Exemplos Rápidos de Integração

Exemplo de verificação de licença via endpoint POST /api/v1/licenses/verify: 

const response = await fetch('http://localhost:3000/api/v1/licenses/verify', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer sk_live_SUA_CHAVE_AQUI'
  },
  body: JSON.stringify({
    licenseKey: 'EJUM-YHEM-UC96-J1VD',
    extensionId: 'dszap'
  })
});

const data = await response.json();
console.log(data);
import requests

url = "http://localhost:3000/api/v1/licenses/verify"
headers = {
    "Content-Type": "application/json",
    "Authorization": "Bearer sk_live_SUA_CHAVE_AQUI"
}
payload = {
    "licenseKey": "EJUM-YHEM-UC96-J1VD",
    "extensionId": "dszap"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
<?php
$ch = curl_init('http://localhost:3000/api/v1/licenses/verify');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
  'Content-Type: application/json',
  'Authorization: Bearer sk_live_SUA_CHAVE_AQUI'
]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
  'licenseKey' => 'EJUM-YHEM-UC96-J1VD',
  'extensionId' => 'dszap'
]));

$response = curl_exec($ch);
curl_close($ch);
echo $response;
?>
